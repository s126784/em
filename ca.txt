## бикластеры

library(biclust)

# Загрузим данные
data(mtcars)

# Посмотрим на первые строки данных
head(mtcars)

# Создадим бинарную матрицу
# Преобразуем значения в 1, если они выше среднего, и 0 если ниже
binary_mtcars <- apply(mtcars, 2, function(x) as.numeric(x > mean(x)))

# Посмотрим на полученную бинарную матрицу
print("Бинарная матрица (первые 5 строк):")
print(head(binary_mtcars, 5))

# Применим Bimax алгоритм
result <- biclust(binary_mtcars, method=BCBimax(), 
                  minr=3,    # минимум 3 машины
                  minc=2,    # минимум 2 характеристики
                  number=5)  # максимум 5 бикластеров

# Выведем результаты
print("\nНайденные бикластеры:")
print(result)

# Подробная информация о первом бикластере
print("\nПервый бикластер:")
bicluster(binary_mtcars, result, number=1)
